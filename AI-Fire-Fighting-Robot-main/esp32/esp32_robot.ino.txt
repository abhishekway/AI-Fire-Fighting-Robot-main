#define ENA 14
#define IN1 27
#define IN2 26

#define ENB 25
#define IN3 33
#define IN4 32

#define RELAY 23

void setup()
{
    Serial.begin(9600);

    pinMode(ENA, OUTPUT);
    pinMode(IN1, OUTPUT);
    pinMode(IN2, OUTPUT);

    pinMode(ENB, OUTPUT);
    pinMode(IN3, OUTPUT);
    pinMode(IN4, OUTPUT);

    pinMode(RELAY, OUTPUT);

    digitalWrite(RELAY, LOW);

    stopRobot();
}

void loop()
{
    if (Serial.available())
    {
        char command = Serial.read();

        if (command == 'F')
        {
            moveForward();

            delay(3000);

            stopRobot();

            digitalWrite(RELAY, HIGH);

            delay(5000);

            digitalWrite(RELAY, LOW);
        }

        else if (command == 'N')
        {
            stopRobot();
            digitalWrite(RELAY, LOW);
        }
    }
}

void moveForward()
{
    digitalWrite(IN1, HIGH);
    digitalWrite(IN2, LOW);

    digitalWrite(IN3, HIGH);
    digitalWrite(IN4, LOW);

    analogWrite(ENA, 200);
    analogWrite(ENB, 200);
}

void stopRobot()
{
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, LOW);

    digitalWrite(IN3, LOW);
    digitalWrite(IN4, LOW);

    analogWrite(ENA, 0);
    analogWrite(ENB, 0);
}
